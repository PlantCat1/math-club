\documentclass[tikz,border=8pt]{standalone}
\usepackage{amsmath}
\usetikzlibrary{angles,quotes,calc,arrows.meta,decorations.pathreplacing}

\begin{document}
\begin{tikzpicture}[
  >=Stealth,
  font=\small,
  pt/.style={circle,fill=black,inner sep=1.3pt},
  ray/.style={black!75,line width=0.7pt},
  chord/.style={red!70!black,line width=1.4pt},
  bisect/.style={green!45!black,line width=0.9pt},
  guide/.style={black!55,dashed,line width=0.6pt},
  ang/.style={draw,angle eccentricity=1.28,line width=0.6pt,
              pic text options={font=\footnotesize}},
]

% ---------- parameters (illustrative angles; the argument is general) ----------
\def\R{6}        % drawn radius (represents 10 ft)
\def\th{40}      % theta
\def\h{32}       % step h
\pgfmathsetmacro{\thh}{\th+\h}        % theta + h
\pgfmathsetmacro{\thm}{\th+\h/2}      % theta + h/2
\pgfmathsetmacro{\Rm}{\R*cos(\h/2)}   % |OM|
\pgfmathsetmacro{\yA}{-\R*cos(\th)}

% ---------- points (angles measured from the downward vertical) ----------
\coordinate (O) at (0,0);
\coordinate (P) at ({-90+\th}:\R);
\coordinate (Q) at ({-90+\thh}:\R);
\coordinate (M) at ({-90+\thm}:\Rm);
\coordinate (A) at (0,\yA);
\coordinate (C) at (Q |- P);

% ---------- fills ----------
\fill[blue!9]   (O) -- (A) -- (P) -- cycle;
\fill[orange!18] (P) -- (C) -- (Q) -- cycle;

% ---------- circle (path of the foot of the ladder) ----------
\draw[black!35,line width=0.7pt]
  (-90:\R) arc[start angle=-90,end angle=18,radius=\R];
\node[black!55,font=\footnotesize,anchor=west] at (16:\R) {circle, radius $10$};

% ---------- vertical reference line through O ----------
\draw[guide] (0,0.9) -- (0,-\R-0.4);

% ---------- right triangle OAP ----------
\draw[ray] (O) -- (P);
\draw[ray] (A) -- (P);
\draw[ray] (O) -- (A);
\draw (P) pic[draw,black,angle radius=3mm] {right angle = P--A--O};

% ---------- radius OQ, chord PQ, perpendicular bisector OM ----------
\draw[ray] (O) -- (Q);
\draw[chord]  (P) -- (Q);
\draw[bisect] (O) -- (M);
\draw (M) pic[draw,green!45!black,angle radius=2.6mm] {right angle = Q--M--O};

% arc PQ emphasised
\draw[black!85,line width=1.4pt]
  ({-90+\th}:\R) arc[start angle={-90+\th},end angle={-90+\thh},radius=\R];

% equal-half ticks on PM and MQ
\path (P) -- (M) coordinate[pos=0.5] (t1);
\path (M) -- (Q) coordinate[pos=0.5] (t2);
\foreach \t in {t1,t2}{
  \draw[chord,line width=0.9pt,rotate around={\thm+90:(\t)}] ($(\t)+(-0.09,0)$) -- ($(\t)+(0.09,0)$);
}

% ---------- horizontal / vertical legs of the projection triangle ----------
\draw[line width=0.9pt,orange!70!black] (P) -- (C);
\draw[line width=0.9pt,orange!70!black] (C) -- (Q);
\draw (C) pic[draw,black,angle radius=2.6mm] {right angle = Q--C--P};

% extend the horizontal through A and P to show x
\draw[guide] (A) -- (P);

% ---------- angle marks ----------
\pic[ang,``$\theta$'',angle radius=11mm]            {angle = A--O--P};
\pic[ang,angle radius=26mm]                       {angle = P--O--Q};
\node[font=\footnotesize] at ({-90+\th+0.78*\h}:3.0) {$h$};
\pic[ang,green!45!black,densely dashed,
     ``$\theta+\tfrac h2$'',angle radius=38mm,
     angle eccentricity=1.12]                     {angle = A--O--M};
\pic[ang,green!45!black,densely dashed,
     ``$\theta{+}\tfrac h2$'',angle radius=7mm,
     angle eccentricity=1.75,pic text options={font=\scriptsize}] {angle = C--P--Q};

% ---------- braces for horizontal distances ----------
\draw[decorate,decoration={brace,mirror,amplitude=5pt}]
  (A) -- (P) node[midway,below=7pt] {$x=10\sin\theta$};
\draw[decorate,decoration={brace,mirror,amplitude=5pt},orange!70!black]
  (P) -- (C) node[midway,below=7pt,text=black] {$\Delta x$};

% ---------- labels ----------
\path (O) -- (P) node[midway,below left=1pt,sloped,black!80] {$10$};
\path (O) -- (Q) node[midway,above=1pt,sloped,black!80] {$10$};
\path (P) -- (Q) node[pos=0.22,above=1pt,sloped,red!70!black] {$PQ$};
\node[right=1pt,orange!70!black] at ($(Q)!0.5!(C)$) {$\Delta y$};

\foreach \p in {O,A,P,Q,M,C} \node[pt] at (\p) {};
\node[above left]   at (O) {$O$};
\node[left]         at (A) {$A$};
\node[below right]  at (P) {$P$};
\node[above right]  at (Q) {$Q$};
\node[below right=0pt,green!45!black] at (M) {$M$};
\node[below right]  at (C) {$C$};

% ---------- summary box ----------
\node[draw=black!40,rounded corners=3pt,fill=black!3,inner sep=7pt,
      anchor=north west,text width=6.3cm,align=left]
  at (8.2,1.2) {%
  \textbf{Construction}\\[3pt]
  \textbf{1.} Right triangle $OAP$: $x=AP=10\sin\theta$.\\[3pt]
  \textbf{2.} Rotate the ladder by $h$: foot moves $P\to Q$ along the arc (dark).
     Chord $PQ=2\cdot10\sin\tfrac h2$.\\[3pt]
  \textbf{3.} $OM\perp PQ$ bisects angle $h$, so $\angle(OA,OM)=\theta+\tfrac h2$.
     Since $PQ\perp OM$, chord $PQ$ makes angle $\theta+\tfrac h2$ with the horizontal $PC$.\\[3pt]
  \textbf{4.} Right triangle $PCQ$:
  \[\Delta x = PQ\cos\!\big(\theta+\tfrac h2\big)=20\sin\tfrac h2\cos\!\big(\theta+\tfrac h2\big)\]
  \textbf{5.} Secant slope:
  \[\frac{\Delta x}{h}=10\cos\!\big(\theta+\tfrac h2\big)\,\frac{\sin(h/2)}{h/2}\]
};

\end{tikzpicture}
\end{document}
